/* RENDER VIDEO CARDS */

const grid = document.getElementById("videoGrid");
const searchInput = document.getElementById("searchInput");

let currentCategory = "All";
let currentNav = "Home";

/* Randomly selected internet images */

function thumbnailUrl(photo) {
  return `https://images.unsplash.com/${photo}?auto=format&fit=crop&w=640&h=360&q=80`;
}

function renderVideos() {

  const query = searchInput.value.trim().toLowerCase();

  let filtered = videos.filter(v => {

    const matchesQuery =
      `${v.title} ${v.channel} ${v.category}`
      .toLowerCase()
      .includes(query);

    const matchesCategory =
      currentCategory === "All" ||
      v.category === currentCategory ||
      (
        currentCategory === "Recently uploaded" &&
        ["1 day ago", "2 days ago", "3 hours ago"].includes(v.age)
      ) ||
      (
        currentCategory === "Watched" &&
        v.title.toLowerCase().includes("habits")
      );

    const matchesNav =
      !["Music", "Gaming", "News", "Trending"].includes(currentNav) ||
      currentNav === v.category ||
      (currentNav === "Trending" && parseFloat(v.views) > 1);

    return matchesQuery && matchesCategory && matchesNav;

  });

  if (currentNav === "Shorts") {
    filtered = filtered.filter(v => v.duration !== "LIVE");
  }

  grid.innerHTML = "";

  if (filtered.length === 0) {
    grid.innerHTML =
      `<div class="empty">No videos found. Try another search or category.</div>`;
    return;
  }

  filtered.forEach(v => {

    const card = document.createElement("article");
    card.className = "video-card";

    card.innerHTML = `
      <div class="thumbnail">
        <img
          src="${thumbnailUrl(v.img)}"
          alt="${v.title}"
          loading="lazy"
          onerror="this.onerror=null;this.src='https://picsum.photos/seed/${encodeURIComponent(v.title)}/640/360';"
        >
        <span class="duration">${v.duration}</span>
      </div>

      <div class="video-info">

        <div class="channel-avatar" style="background:${v.color}">
          ${v.initial}
        </div>

        <div class="video-text">
          <h3 class="video-title">${v.title}</h3>
          <div class="meta">
            ${v.channel}<br>
            ${v.views} · ${v.age}
          </div>
        </div>

        <button class="dots" aria-label="More options">⋮</button>

      </div>
    `;

    card.querySelector(".thumbnail")
      .addEventListener("click", () => openPlayer(v));

    card.querySelector(".video-title")
      .addEventListener("click", () => openPlayer(v));

    card.querySelector(".dots")
      .addEventListener("click", e => {
        e.stopPropagation();
        alert("More options for: " + v.title);
      });

    grid.appendChild(card);

  });

}

/* VIDEO PREVIEW */

function openPlayer(video) {

  document.getElementById("playerTitle").textContent = video.title;

  document.getElementById("playerMessage").textContent =
    "▶ " + video.title;

  document.getElementById("playerModal").classList.add("open");

}

document.getElementById("closePlayer")
  .addEventListener("click", () => {
    document.getElementById("playerModal").classList.remove("open");
  });

document.getElementById("playerModal")
  .addEventListener("click", e => {
    if (e.target.id === "playerModal") {
      e.currentTarget.classList.remove("open");
    }
  });

document.addEventListener("keydown", e => {
  if (e.key === "Escape") {
    document.getElementById("playerModal").classList.remove("open");
  }
});

    document.querySelectorAll(".side-link")
      .forEach(el => {
        el.classList.toggle("active", el === link);
      });

    currentNav = link.dataset.nav;

    currentCategory =
      ["Music", "Gaming", "News"].includes(currentNav)
        ? currentNav
        : "All";

    document.querySelectorAll(".chips button")
      .forEach(b => {
        b.classList.toggle(
          "selected",
          b.dataset.category === currentCategory
        );
      });

    renderVideos();

  });

    const SpeechRecognition =
      window.SpeechRecognition || window.webkitSpeechRecognition;

    const recognition = new SpeechRecognition();

    recognition.lang = "en-IN";

    recognition.onresult = e => {
      searchInput.value = e.results[0][0].transcript;
      renderVideos();
    };

    recognition.start();

  });
